백엔드: NestJS + Prisma

NestJS 모듈 기반 구조로 64개 비즈니스 모듈을 관리합니다. 각 모듈은 4개 파일(module, controller, service, spec)로 구성됩니다.

// codes.controller.ts
@Controller('codes')
export class CodesController {
  constructor(private readonly codesService: CodesService) {}

  @Get()
  findAll(@Query('search') search?: string) {
    return this.codesService.findAll(search);
  }
}

// codes.service.ts — 트랜잭션 패턴
@Injectable()
export class CodesService {
  constructor(private readonly prisma: PrismaService) {}

  async create(dto: CreateCodeDto, userId: string) {
    const existing = await this.prisma.codeMaster.findUnique({
      where: { codeType: dto.codeType },
    });
    if (existing) throw new ConflictException('이미 존재하는 코드그룹입니다');

    const { details, ...masterData } = dto;

    // 마스터 + 디테일 원자적 생성
    return this.prisma.$transaction(async (tx) => {
      await tx.codeMaster.create({
        data: { ...masterData, createdBy: userId, updatedBy: userId },
      });
      if (details.length > 0) {
        await tx.codeDetail.createMany({
          data: details.map((d) => ({
            codeType: dto.codeType, ...d,
            createdBy: userId, updatedBy: userId,
          })),
        });
      }
    });
  }
}

UI 컨벤션: AS-IS / TO-BE 탭

모든 업무 화면은 두 탭으로 구성됩니다. 기존 사용자가 레거시와 신규 화면을 나란히 비교·검증할 수 있습니다.

공통코드관리
AS-IS
TO-BE
AS-IS 탭
레거시 .NET WinForms 화면을 웹에서 그대로 재현. 기존 사용자 검증 및 마이그레이션 비교용.
TO-BE 탭 (기본)
UI/UX를 개선한 신규 화면. 최종 운영 대상. useState('TO-BE')로 기본 활성화.

페이지 컴포넌트 표준 패턴 (CommonCodePage.tsx 기준):

type TabType = 'AS-IS' | 'TO-BE';

export default function CommonCodePage() {
  const [activeTab, setActiveTab] = useState<TabType>('TO-BE'); // TO-BE가 기본
  const { isDirty, setDirty } = useCodeStore();

  const handleTabChange = useCallback(
    (tab: TabType) => {
      // dirty-state: 저장하지 않은 변경이 있으면 확인
      if (isDirty) {
        const confirmed = window.confirm('저장하지 않은 변경사항이 있습니다. 이동하시겠습니까?');
        if (!confirmed) return;
        setDirty(false);
      }
      setActiveTab(tab);
    },
    [isDirty, setDirty]
  );

  return (
    <div className="h-full flex flex-col">
      <div className="flex items-center justify-between mb-4">
        <h2 className="text-lg font-semibold">공통코드관리</h2>
        <div className="flex rounded-lg border border-gray-300">
          {(['AS-IS', 'TO-BE'] as TabType[]).map((tab) => (
            <button
              key={tab}
              onClick={() => handleTabChange(tab)}
              className={activeTab === tab ? 'bg-blue-600 text-white' : 'bg-white text-gray-600'}
            >
              {tab}
            </button>
          ))}
        </div>
      </div>
      <div className="flex-1 overflow-hidden">
        {activeTab === 'AS-IS' ? <AsIsTab /> : <ToBeTab />}
      </div>
    </div>
  );
}

Zustand 스토어 패턴 — 모든 모듈에서 isDirty / setDirty가 필수입니다:

// shared/stores/code-store.ts
interface CodeState {
  masters: CodeMasterSummary[];
  isLoading: boolean;
  isDirty: boolean;           // 필수: 탭 전환 시 dirty-state 체크
  loadMasters: (search?: string) => Promise<void>;
  setDirty: (dirty: boolean) => void;  // 필수
}

export const useCodeStore = create<CodeState>((set) => ({
  masters: [],
  isDirty: false,
  loadMasters: async (search?: string) => {
    set({ isLoading: true });
    try {
      const { data } = await api.get('/codes', { params: { search } });
      set({ masters: data, isLoading: false });
    } catch (err: any) {
      set({ error: err?.response?.data?.message, isLoading: false });
    }
  },
  setDirty: (dirty) => set({ isDirty: dirty }),
}));

변환 예시 (Oracle SP → Prisma):

-- Oracle SP_SAVE_EQUIP (레거시)
PROCEDURE SP_SAVE_EQUIP(p_equip_cd VARCHAR2, p_equip_nm VARCHAR2) IS
BEGIN
  SELECT COUNT(*) INTO v_cnt FROM TB_EQUIP WHERE EQUIP_CD = p_equip_cd;
  IF v_cnt > 0 THEN
    RAISE_APPLICATION_ERROR(-20001, '이미 등록된 설비코드입니다.');
  END IF;
  INSERT INTO TB_EQUIP (EQUIP_CD, EQUIP_NM, REG_DT)
    VALUES (p_equip_cd, p_equip_nm, SYSDATE);
  COMMIT;
END;
// NestJS 변환 후
async create(dto: CreateEquipDto, userId: string) {
  const existing = await this.prisma.equipment.findUnique({
    where: { equipCd: dto.equipCd },
  });
  if (existing) throw new ConflictException('이미 등록된 설비코드입니다.');

  return this.prisma.equipment.create({
    data: { ...dto, createdBy: userId, updatedBy: userId },
  });
}

Oracle → PostgreSQL 데이터 마이그레이션

각 화면 구현 후 Python 스크립트로 실데이터를 이전합니다.

# prisma/migrate-equipment-direct.py (패턴 예시)
import cx_Oracle, psycopg2

def migrate_equipment():
    # 1. Oracle에서 데이터 추출
    oracle_conn = cx_Oracle.connect(ORACLE_DSN)
    cursor = oracle_conn.cursor()
    cursor.execute("""
        SELECT EQUIP_CD, EQUIP_NM, EQUIP_TYPE, REG_DT
        FROM TB_EQUIP WHERE USE_YN = 'Y' ORDER BY EQUIP_CD
    """)
    rows = cursor.fetchall()

    # 2. PostgreSQL에 적재 (트랜잭션)
    pg_conn = psycopg2.connect(PG_DSN)
    with pg_conn.cursor() as pg_cur:
        pg_cur.executemany("""
            INSERT INTO "Equipment" (equip_cd, equip_nm, equip_type, created_at)
            VALUES (%s, %s, %s, %s) ON CONFLICT (equip_cd) DO NOTHING
        """, rows)
    pg_conn.commit()

    # 3. 건수 검증
    print(f"마이그레이션 완료: {len(rows)}건")

// codes.spec.ts — Red: 먼저 실패하는 테스트 작성
describe('CodesService', () => {
  describe('create', () => {
    it('중복 코드그룹 생성 시 ConflictException을 던진다', async () => {
      // Arrange: 이미 존재하는 코드그룹 mock
      mockPrisma.codeMaster.findUnique.mockResolvedValue({ codeType: 'MAT' });

      // Act & Assert
      await expect(
        service.create({ codeType: 'MAT', codeTypeName: '재질', details: [] }, 'user1')
      ).rejects.toThrow(ConflictException);
    });

    it('신규 코드그룹 생성 시 트랜잭션으로 마스터+디테일을 저장한다', async () => {
      mockPrisma.codeMaster.findUnique.mockResolvedValue(null);
      mockPrisma.$transaction.mockImplementation(async (fn) => fn(mockPrisma));

      await service.create({
        codeType: 'NEW', codeTypeName: '신규',
        details: [{ code: '01', codeName: '항목1' }],
      }, 'user1');

      expect(mockPrisma.codeMaster.create).toHaveBeenCalledWith(
        expect.objectContaining({ data: expect.objectContaining({ codeType: 'NEW' }) })
      );
      expect(mockPrisma.codeDetail.createMany).toHaveBeenCalled();
    });
  });
});

Tidy First: 구조적/행위적 변경 분리

Ward Cunningham의 "Tidy First" 원칙을 커밋 단위로 적용합니다.

STRUCTURAL
동작이 바뀌지 않는다
  • 이름 변경, 파일 이동
  • 메서드 추출, 포맷팅
  • 커밋: [STRUCTURAL] refactor: ...
BEHAVIORAL
동작이 바뀐다
  • 새 기능 추가, 버그 수정
  • 로직 변경, 새 테스트
  • 커밋: [BEHAVIORAL] feat: ...

실제 커밋 이력 예시:

edfdac0 [STRUCTURAL] refactor: Prisma 2개 모델 추가 (MeltResult, EquipmentOperation)
eb42222 [BEHAVIORAL] feat: PR04010 용해 실적 화면 구현
9d519b4 [BEHAVIORAL] feat: PR03014 공정별 작업조정 화면 구현

절대로 하지 않는 것: 리팩토링과 기능 추가를 같은 커밋에 섞는 것 · 테스트 실패 상태에서 커밋하는 것 · git add .로 전체 스테이징하는 것

# 1. 저장소 클론
git clone <repository-url>
cd seah-cpe

# 2. 의존성 설치 (Node 20+, pnpm 10+ 필요)
pnpm install

# 3. 환경변수 설정
# apps/api-server/.env
DATABASE_URL="postgresql://user:password@localhost:5432/seah_cpe"
JWT_SECRET="your-secret-key-here"
JWT_EXPIRES_IN="7d"

# apps/user-client/.env.local
VITE_API_URL="http://localhost:3001"

# 4. DB 초기화
cd apps/api-server
pnpm prisma db push        # 스키마 적용 (80개 모델)
pnpm prisma generate       # Prisma 클라이언트 생성

# 5. 개발 서버 시작
cd ../..
pnpm dev                   # API 3001 + Client 5174 동시 시작

? 부록: 자주 묻는 질문

AS-IS 탭은 언제까지 유지하나요?
실제 사용자가 TO-BE 탭을 검증·승인하면 AS-IS 탭을 제거합니다. 마이그레이션 기간 동안 병행 운영합니다.
새 모듈 추가 시 어떤 파일을 만드나요?
최소 4개: {name}.module.ts, {name}.service.ts, {name}.controller.ts, {name}.spec.ts. NestJS CLI: nest g module {name}으로 스캐폴딩할 수 있습니다.
Prisma 스키마 변경 시 어떻게 하나요?
prisma/schema.prisma 수정 → pnpm prisma db push → pnpm prisma generate. 마이그레이션 파일이 아닌 db push 방식을 사용합니다.
테스트가 실패하면 어떻게 하나요?
절대 커밋하지 않습니다. 실패 원인을 찾아 수정하거나, 스펙이 변경됐으면 테스트도 함께 수정합니다.
Oracle 데이터 마이그레이션은 언제 하나요?
화면 구현이 완료된 후, 운영 전환 직전에 실행합니다. 개발 중에는 시드 데이터나 샘플 데이터를 사용합니다.
