Module 07 — 실습

고급 기능 실습

사전 요구사항 kind, kubectl, helm이 설치되어 있어야 합니다.

실습: StatefulSet 클러스터링 배포

kind create cluster --name advanced-lab

cat > advanced-values.yaml << 'EOF'
alloy:
  configMap:
    content: |
      // 클러스터링 선언
      clustering {
        enabled = true
      }

      discovery.kubernetes "pods" {
        role = "pod"
      }

      // clustering.enabled = true → 타깃 자동 분배
      prometheus.scrape "clustered" {
        targets    = discovery.kubernetes.pods.targets
        clustering {
          enabled = true
        }
        forward_to = [prometheus.remote_write.mimir.receiver]
      }

      prometheus.remote_write "mimir" {
        endpoint {
          url = "http://mimir:9009/api/v1/push"
        }
      }

controller:
  type: statefulset
  replicas: 3

# 클러스터링용 headless service
service:
  clusterIP: None

rbac:
  create: true
serviceAccount:
  create: true

extraArgs:
  - --cluster.enabled
  - --cluster.node-name=$(POD_NAME)
  - --cluster.join-peers=alloy-0.alloy-headless:12345

extraEnv:
  - name: POD_NAME
    valueFrom:
      fieldRef:
        fieldPath: metadata.name
EOF

helm install alloy grafana/alloy   --namespace monitoring --create-namespace   -f advanced-values.yaml

클러스터 상태 확인

kubectl get pods -n monitoring
# 3개 Pod가 Running 상태인지 확인

kubectl port-forward -n monitoring alloy-0 12345:12345 &
# http://localhost:12345 → Cluster 탭에서 멤버 목록 확인

정리

kind delete cluster --name advanced-lab
