Module 04 — 실습

메트릭 수집 실습

사전 요구사항 kind, kubectl, helm이 설치되어 있어야 합니다.

실습 1: Node Exporter 메트릭 수집

kind create cluster --name metrics-lab

# Node Exporter 설치
helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
helm install node-exporter prometheus-community/prometheus-node-exporter   --namespace monitoring --create-namespace

실습 2: Alloy 메트릭 파이프라인 설정

cat > metrics-values.yaml << 'EOF'
alloy:
  configMap:
    content: |
      // Node Exporter 디스커버리
      discovery.kubernetes "nodes" {
        role = "node"
      }

      discovery.relabel "node_exporter" {
        targets = discovery.kubernetes.nodes.targets

        rule {
          source_labels = ["__address__"]
          regex         = "(.+):.*"
          replacement   = "$1:9100"
          target_label  = "__address__"
        }

        rule {
          source_labels = ["__meta_kubernetes_node_name"]
          target_label  = "node"
        }
      }

      prometheus.scrape "node_exporter" {
        targets         = discovery.relabel.node_exporter.output
        scrape_interval = "30s"
        forward_to      = [prometheus.relabel.drop_noisy.receiver]
      }

      // 노이즈 메트릭 필터링
      prometheus.relabel "drop_noisy" {
        forward_to = [prometheus.remote_write.mimir.receiver]

        rule {
          source_labels = ["__name__"]
          action        = "drop"
          regex         = "node_cpu_guest_.*"
        }
      }

      prometheus.remote_write "mimir" {
        endpoint {
          url = "http://mimir-service:9009/api/v1/push"
        }
      }

      // Alloy 자신도 수집
      prometheus.exporter.self "alloy" {}

      prometheus.scrape "alloy_self" {
        targets    = prometheus.exporter.self.alloy.targets
        forward_to = [prometheus.remote_write.mimir.receiver]
      }

controller:
  type: daemonset
rbac:
  create: true
serviceAccount:
  create: true
EOF

helm install alloy grafana/alloy   --namespace monitoring   -f metrics-values.yaml

실습 3: 수집 확인

kubectl port-forward -n monitoring svc/alloy 12345:12345 &

# UI에서 확인
# Components → prometheus.scrape.node_exporter
# → Exports → 최근 스크래핑 상태

# 메트릭 직접 조회
curl http://localhost:12345/metrics | grep alloy_build

정리

kind delete cluster --name metrics-lab
