Module 06 — 실습

트레이스 수집 실습

사전 요구사항 kind, kubectl, helm이 설치되어 있어야 합니다.

실습: OTLP 트레이스 파이프라인 구성

kind create cluster --name traces-lab

cat > traces-values.yaml << 'EOF'
alloy:
  configMap:
    content: |
      // OTLP 수신기
      otelcol.receiver.otlp "default" {
        grpc { endpoint = "0.0.0.0:4317" }
        http { endpoint = "0.0.0.0:4318" }

        output {
          traces = [otelcol.processor.batch.default.input]
        }
      }

      // 배치 처리
      otelcol.processor.batch "default" {
        timeout         = "5s"
        send_batch_size = 1000

        output {
          traces = [otelcol.processor.tail_sampling.smart.input]
        }
      }

      // Tail-based 샘플링
      otelcol.processor.tail_sampling "smart" {
        decision_wait = "10s"

        policy {
          name = "errors"
          type = "status_code"
          status_code {
            status_codes = ["ERROR"]
          }
        }

        policy {
          name = "random-20"
          type = "probabilistic"
          probabilistic {
            sampling_percentage = 20
          }
        }

        output {
          traces = [otelcol.exporter.otlp.tempo.input]
        }
      }

      otelcol.exporter.otlp "tempo" {
        client {
          endpoint = "tempo:4317"
          tls { insecure = true }
        }
      }

service:
  enabled: true
  ports:
    otlp-grpc:
      port: 4317
      targetPort: 4317
      protocol: TCP

controller:
  type: deployment
  replicas: 1
EOF

helm install alloy grafana/alloy   --namespace monitoring --create-namespace   -f traces-values.yaml

테스트 트레이스 전송

kubectl port-forward -n monitoring svc/alloy 4317:4317 &
kubectl port-forward -n monitoring svc/alloy 12345:12345 &

# Python으로 테스트 트레이스 전송 (opentelemetry-sdk 필요)
pip install opentelemetry-sdk opentelemetry-exporter-otlp

python3 << 'PYEOF'
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter

provider = TracerProvider()
exporter = OTLPSpanExporter(endpoint="http://localhost:4317", insecure=True)
provider.add_span_processor(BatchSpanProcessor(exporter))
trace.set_tracer_provider(provider)

tracer = trace.get_tracer("test-tracer")
with tracer.start_as_current_span("test-span") as span:
    span.set_attribute("test.attribute", "hello-alloy")
    print("Trace sent!")
PYEOF
